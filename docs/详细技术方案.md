```python
# src/data_sources/football_data_co_uk.py
import pandas as pd
import requests
from pathlib import Path

class FootballDataUK:
    """football-data.co.uk 免费CSV数据源"""
    
    BASE_URL = "https://www.football-data.co.uk/mmz4281"
    # 联赛代码映射
    LEAGUE_CODES = {
        "英超": "E0", "英冠": "E1",
        "西甲": "SP1", "西乙": "SP2",
        "德甲": "D1", "德乙": "D2",
        "意甲": "I1", "意乙": "I2",
        "法甲": "F1", "法乙": "F2",
    }
    
    def __init__(self, cache_dir="data/raw"):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(parents=True, exist_ok=True)
    
    def fetch_season(self, league: str, season: str = "2627") -> pd.DataFrame:
        """获取指定联赛指定赛季的比赛数据
        season格式: 2627 表示 2026-27 赛季
        """
        code = self.LEAGUE_CODES.get(league)
        if not code:
            raise ValueError(f"不支持的联赛: {league}")
        
        url = f"{self.BASE_URL}/{season}/{code}.csv"
        cache_file = self.cache_dir / f"{code}_{season}.csv"
        
        # 优先读缓存
        if cache_file.exists():
            return pd.read_csv(cache_file, encoding="utf-8-sig")
        
        resp = requests.get(url, timeout=30)
        resp.raise_for_status()
        cache_file.write_bytes(resp.content)
        
        df = pd.read_csv(cache_file, encoding="utf-8-sig")
        return self._clean_data(df, league, season)
    
    def fetch_multiple_seasons(self, league: str, seasons: list) -> pd.DataFrame:
        """获取多个赛季数据（用于模型训练）"""
        frames = [self.fetch_season(league, s) for s in seasons]
        return pd.concat(frames, ignore_index=True)
    
    def _clean_data(self, df: pd.DataFrame, league: str, season: str) -> pd.DataFrame:
        """数据清洗：统一字段名、处理缺失值"""
        # 保留核心字段
        core_cols = ["Div","Date","Time","HomeTeam","AwayTeam",
                      "FTHG","FTAG","FTR","HTHG","HTAG","HTR",
                      "HS","AS","HST","AST","HF","AF","HC","AC",
                      "HY","AY","HR","AR"]
        # xG字段（新赛季有）
        xg_cols = [c for c in df.columns if "xG" in c or "xg" in c.lower()]
        cols = [c for c in core_cols + xg_cols if c in df.columns]
        df = df[cols].copy()
        
        # 统一日期格式
        df["Date"] = pd.to_datetime(df["Date"], format="%d/%m/%Y", errors="coerce")
        df["league"] = league
        df["season"] = season
        return df
```

```python
import requests
from typing import Dict, List, Optional

class ApiFootball:
    """API-Football (RapidAPI) 实时数据源
    免费层: 100 requests/day
    文档: https://www.api-football.com/documentation-v3
    """
    
    BASE_URL = "https://api-football-v1.p.rapidapi.com/v3"
    
    def __init__(self, api_key: str, host: str = "api-football-v1.p.rapidapi.com"):
        self.headers = {
            "X-RapidAPI-Key": api_key,
            "X-RapidAPI-Host": host,
        }
    
    def get_live_scores(self) -> List[Dict]:
        """获取实时比分"""
        resp = requests.get(f"{self.BASE_URL}/fixtures", 
                           headers=self.headers, params={"live": "all"})
        return resp.json().get("response", [])
    
    def get_team_squad(self, team_id: int) -> List[Dict]:
        """获取球队阵容"""
        resp = requests.get(f"{self.BASE_URL}/players/squads",
                           headers=self.headers, params={"team": team_id})
        return resp.json().get("response", [])
    
    def get_player_stats(self, player_id: int, season: int) -> Dict:
        """获取球员赛季统计"""
        resp = requests.get(f"{self.BASE_URL}/players",
                           headers=self.headers, 
                           params={"id": player_id, "season": season})
        return resp.json().get("response", [])
    
    def get_transfers(self, team_id: Optional[int] = None, 
                      player_id: Optional[int] = None) -> List[Dict]:
        """获取转会信息"""
        params = {}
        if team_id: params["team"] = team_id
        if player_id: params["player"] = player_id
        resp = requests.get(f"{self.BASE_URL}/transfers",
                           headers=self.headers, params=params)
        return resp.json().get("response", [])
    
    def get_fixtures_by_date(self, date: str) -> List[Dict]:
        """获取指定日期的赛程 (date格式: YYYY-MM-DD)"""
        resp = requests.get(f"{self.BASE_URL}/fixtures",
                           headers=self.headers, params={"date": date})
        return resp.json().get("response", [])
```

### 4.1 Elo 评级系统

Elo 是衡量球队实力的核心指标，也是预测模型最重要的特征之一。

```python
# src/analysis/elo_rating.py
class EloRating:
    """球队Elo评级系统
    基于国际象棋Elo算法，适配足球比赛
    """
    
    def __init__(self, initial_rating: float = 1500.0, 
                 home_advantage: float = 65.0,
                 k_factor: float = 32.0):
        self.ratings = {}  # team_id -> rating
        self.initial_rating = initial_rating
        self.home_advantage = home_advantage  # 主场优势加成
        self.k_factor = k_factor  # 调整系数
    
    def get_rating(self, team_id: int) -> float:
        return self.ratings.get(team_id, self.initial_rating)
    
    def expected_score(self, rating_a: float, rating_b: float, 
                       is_home_a: bool = True) -> float:
        """计算A队预期胜率"""
        ra = rating_a + (self.home_advantage if is_home_a else 0)
        rb = rating_b
        return 1.0 / (1.0 + 10 ** ((rb - ra) / 400.0))
    
    def update_after_match(self, home_id: int, away_id: int,
                           home_goals: int, away_goals: int):
        """比赛后更新Elo评级"""
        # 实际结果
        if home_goals > away_goals:
            actual_home, actual_away = 1.0, 0.0
        elif home_goals < away_goals:
            actual_home, actual_away = 0.0, 1.0
        else:
            actual_home = actual_away = 0.5
        
        # 预期结果
        exp_home = self.expected_score(self.get_rating(home_id),
                                        self.get_rating(away_id), True)
        exp_away = 1.0 - exp_home
        
        # 净胜球加成（赢越多，K越大）
        goal_diff = abs(home_goals - away_goals)
        k_multiplier = 1.0 + 0.1 * goal_diff  # 每多进1球，K增加10%
        
        # 更新
        self.ratings[home_id] = self.get_rating(home_id) + \
            self.k_factor * k_multiplier * (actual_home - exp_home)
        self.ratings[away_id] = self.get_rating(away_id) + \
            self.k_factor * k_multiplier * (actual_away - exp_away)
    
    def batch_update(self, matches_df):
        """批量更新（按时间顺序）"""
        df = matches_df.sort_values("Date")
        for _, row in df.iterrows():
            if pd.notna(row.get("FTHG")):
                self.update_after_match(
                    row["HomeTeam"], row["AwayTeam"],
                    int(row["FTHG"]), int(row["FTAG"])
                )
```

### 4.2 球队统计分析

```python
# src/analysis/team_stats.py
import pandas as pd
import numpy as np

class TeamStats:
    """球队近期状态统计"""
    
    def __init__(self, matches_df: pd.DataFrame):
        self.df = matches_df
    
    def recent_form(self, team: str, n: int = 5) -> dict:
        """计算球队近n场状态
        返回: 胜/平/负, 进球, 失球, 积分, 场均xG
        """
        home = self.df[self.df["HomeTeam"] == team].copy()
        away = self.df[self.df["AwayTeam"] == team].copy()
        
        home["team_goals"] = home["FTHG"]
        home["opp_goals"] = home["FTAG"]
        home["result"] = home["FTR"].map({"H": "W", "D": "D", "A": "L"})
        
        away["team_goals"] = away["FTAG"]
        away["opp_goals"] = away["FTHG"]
        away["result"] = away["FTR"].map({"H": "L", "D": "D", "A": "W"})
        
        all_matches = pd.concat([home, away]).sort_values("Date", ascending=False)
        recent = all_matches.head(n)
        
        points = recent["result"].map({"W": 3, "D": 1, "L": 0}).sum()
        return {
            "team": team,
            "matches": len(recent),
            "wins": (recent["result"] == "W").sum(),
            "draws": (recent["result"] == "D").sum(),
            "losses": (recent["result"] == "L").sum(),
            "goals_for": recent["team_goals"].sum(),
            "goals_against": recent["opp_goals"].sum(),
            "points": points,
            "points_per_game": round(points / len(recent), 2) if len(recent) > 0 else 0,
            "avg_goals_for": round(recent["team_goals"].mean(), 2),
            "avg_goals_against": round(recent["opp_goals"].mean(), 2),
        }
    
    def home_away_record(self, team: str) -> dict:
        """主客场战绩"""
        home = self.df[self.df["HomeTeam"] == team]
        away = self.df[self.df["AwayTeam"] == team]
        return {
            "home_wins": (home["FTR"] == "H").sum(),
            "home_draws": (home["FTR"] == "D").sum(),
            "home_losses": (home["FTR"] == "A").sum(),
            "away_wins": (away["FTR"] == "A").sum(),
            "away_draws": (away["FTR"] == "D").sum(),
            "away_losses": (away["FTR"] == "H").sum(),
        }
    
    def head_to_head(self, team_a: str, team_b: str, n: int = 10) -> dict:
        """历史交锋记录"""
        h2h = self.df[
            ((self.df["HomeTeam"] == team_a) & (self.df["AwayTeam"] == team_b)) |
            ((self.df["HomeTeam"] == team_b) & (self.df["AwayTeam"] == team_a))
        ].sort_values("Date", ascending=False).head(n)
        
        a_wins = ((h2h["HomeTeam"] == team_a) & (h2h["FTR"] == "H")).sum() + \
                 ((h2h["AwayTeam"] == team_a) & (h2h["FTR"] == "A")).sum()
        b_wins = ((h2h["HomeTeam"] == team_b) & (h2h["FTR"] == "H")).sum() + \
                 ((h2h["AwayTeam"] == team_b) & (h2h["FTR"] == "A")).sum()
        draws = (h2h["FTR"] == "D").sum()
        
        return {"team_a_wins": a_wins, "team_b_wins": b_wins, 
                "draws": draws, "total": len(h2h)}
```

### 5.1 特征工程（最重要的部分）

预测模型的效果 80% 取决于特征工程。以下是完整的特征体系：

```python
# src/prediction/feature_engineer.py
import pandas as pd
import numpy as np
from src.analysis.elo_rating import EloRating
from src.analysis.team_stats import TeamStats

class FeatureEngineer:
    """比赛预测特征工程
    特征分为 6 大类，共约 40+ 维特征
    """
    
    def __init__(self, matches_df: pd.DataFrame, elo: EloRating):
        self.df = matches_df
        self.elo = elo
        self.stats = TeamStats(matches_df)
    
    def build_features(self, home_team: str, away_team: str,
                       match_date=None) -> dict:
        """为一场比赛构建完整特征向量"""
        features = {}
        
        # === 1. Elo实力特征 (5维) ===
        home_elo = self.elo.get_rating(home_team)
        away_elo = self.elo.get_rating(away_team)
        features["home_elo"] = home_elo
        features["away_elo"] = away_elo
        features["elo_diff"] = home_elo - away_elo
        features["elo_expected_home_win"] = self.elo.expected_score(home_elo, away_elo, True)
        features["home_elo_advantage"] = home_elo + self.elo.home_advantage - away_elo
        
        # === 2. 近期状态特征 (12维) ===
        home_form_5 = self.stats.recent_form(home_team, 5)
        away_form_5 = self.stats.recent_form(away_team, 5)
        home_form_10 = self.stats.recent_form(home_team, 10)
        away_form_10 = self.stats.recent_form(away_team, 10)
        
        features["home_ppg_5"] = home_form_5["points_per_game"]
        features["away_ppg_5"] = away_form_5["points_per_game"]
        features["home_ppg_10"] = home_form_10["points_per_game"]
        features["away_ppg_10"] = away_form_10["points_per_game"]
        features["home_goals_for_5"] = home_form_5["avg_goals_for"]
        features["away_goals_for_5"] = away_form_5["avg_goals_for"]
        features["home_goals_against_5"] = home_form_5["avg_goals_against"]
        features["away_goals_against_5"] = away_form_5["avg_goals_against"]
        features["home_wins_5"] = home_form_5["wins"]
        features["away_wins_5"] = away_form_5["wins"]
        features["form_diff_ppg_5"] = home_form_5["points_per_game"] - away_form_5["points_per_game"]
        features["form_diff_goals_5"] = home_form_5["avg_goals_for"] - away_form_5["avg_goals_for"]
        
        # === 3. 主客场特征 (6维) ===
        home_ha = self.stats.home_away_record(home_team)
        away_ha = self.stats.home_away_record(away_team)
        features["home_home_win_rate"] = home_ha["home_wins"] / max(home_ha["home_wins"] + home_ha["home_draws"] + home_ha["home_losses"], 1)
        features["away_away_win_rate"] = away_ha["away_wins"] / max(away_ha["away_wins"] + away_ha["away_draws"] + away_ha["away_losses"], 1)
        features["home_home_goals_for_avg"] = self._team_home_goals(home_team, "for")
        features["home_home_goals_against_avg"] = self._team_home_goals(home_team, "against")
        features["away_away_goals_for_avg"] = self._team_away_goals(away_team, "for")
        features["away_away_goals_against_avg"] = self._team_away_goals(away_team, "against")
        
        # === 4. 历史交锋特征 (4维) ===
        h2h = self.stats.head_to_head(home_team, away_team, 10)
        features["h2h_home_wins"] = h2h["team_a_wins"]
        features["h2h_away_wins"] = h2h["team_b_wins"]
        features["h2h_draws"] = h2h["draws"]
        features["h2h_home_win_rate"] = h2h["team_a_wins"] / max(h2h["total"], 1)
        
        # === 5. 进攻防守效率特征 (6维) ===
        features["home_attack_strength"] = self._attack_strength(home_team)
        features["away_attack_strength"] = self._attack_strength(away_team)
        features["home_defense_strength"] = self._defense_strength(home_team)
        features["away_defense_strength"] = self._defense_strength(away_team)
        # 预期进球估算
        features["expected_home_goals"] = features["home_attack_strength"] * features["away_defense_strength"]
        features["expected_away_goals"] = features["away_attack_strength"] * features["home_defense_strength"]
        
        # === 6. 联赛/赛季特征 (3维) ===
        features["home_league_position"] = self._current_position(home_team)
        features["away_league_position"] = self._current_position(away_team)
        features["position_diff"] = features["home_league_position"] - features["away_league_position"]
        
        return features
    
    def _attack_strength(self, team: str) -> float:
        """进攻强度 = 球队场均进球 / 联赛场均进球"""
        team_goals = self.df[(self.df["HomeTeam"] == team)]["FTHG"].sum() + \
                     self.df[(self.df["AwayTeam"] == team)]["FTAG"].sum()
        matches = len(self.df[(self.df["HomeTeam"] == team)]) + \
                  len(self.df[(self.df["AwayTeam"] == team)])
        if matches == 0: return 1.0
        team_avg = team_goals / matches
        league_avg = (self.df["FTHG"].sum() + self.df["FTAG"].sum()) / (len(self.df) * 2)
        return team_avg / league_avg if league_avg > 0 else 1.0
    
    def _defense_strength(self, team: str) -> float:
        """防守强度 = 球队场均失球 / 联赛场均失球（越低越好）"""
        team_conceded = self.df[(self.df["HomeTeam"] == team)]["FTAG"].sum() + \
                        self.df[(self.df["AwayTeam"] == team)]["FTHG"].sum()
        matches = len(self.df[(self.df["HomeTeam"] == team)]) + \
                  len(self.df[(self.df["AwayTeam"] == team)])
        if matches == 0: return 1.0
        team_avg = team_conceded / matches
        league_avg = (self.df["FTHG"].sum() + self.df["FTAG"].sum()) / (len(self.df) * 2)
        return team_avg / league_avg if league_avg > 0 else 1.0
```

```python
# src/prediction/model_trainer.py
import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.model_selection import TimeSeriesSplit, cross_val_score
from sklearn.metrics import accuracy_score, log_loss, classification_report
import joblib
from pathlib import Path

class ModelTrainer:
    """比赛结果预测模型训练
    目标: 三分类 (主胜/平局/客胜)
    模型: XGBoost (表格数据SOTA)
    """
    
    def __init__(self, feature_engineer, model_dir="models"):
        self.fe = feature_engineer
        self.model_dir = Path(model_dir)
        self.model_dir.mkdir(parents=True, exist_ok=True)
        self.model = None
    
    def prepare_training_data(self, matches_df: pd.DataFrame) -> tuple:
        """从历史比赛构建训练集
        注意: 必须按时间顺序构建特征，避免数据泄露！
        """
        X_list, y_list = [], []
        
        df_sorted = matches_df.sort_values("Date").reset_index(drop=True)
        
        for idx, row in df_sorted.iterrows():
            if pd.isna(row.get("FTR")):
                continue
            # 只用当前比赛之前的数据构建特征（关键！防止泄露）
            historical_df = df_sorted.iloc[:idx]
            if len(historical_df) < 50:  # 数据太少跳过
                continue
            
            try:
                fe = FeatureEngineer(historical_df, self.fe.elo)
                features = fe.build_features(row["HomeTeam"], row["AwayTeam"])
                X_list.append(features)
                y_list.append({"H": 0, "D": 1, "A": 2}[row["FTR"]])
            except Exception:
                continue
        
        X = pd.DataFrame(X_list)
        y = np.array(y_list)
        return X, y
    
    def train(self, X: pd.DataFrame, y: np.ndarray, 
              test_size: float = 0.2) -> dict:
        """训练模型，时间序列分割（不能随机分割！）"""
        split_idx = int(len(X) * (1 - test_size))
        X_train, X_test = X.iloc[:split_idx], X.iloc[split_idx:]
        y_train, y_test = y[:split_idx], y[split_idx:]
        
        # XGBoost 三分类
        self.model = xgb.XGBClassifier(
            objective="multi:softprob",
            num_class=3,
            n_estimators=300,
            max_depth=5,
            learning_rate=0.05,
            subsample=0.8,
            colsample_bytree=0.8,
            reg_alpha=0.1,
            reg_lambda=1.0,
            random_state=42,
            eval_metric="mlogloss",
            early_stopping_rounds=50,
        )
        
        self.model.fit(
            X_train, y_train,
            eval_set=[(X_test, y_test)],
            verbose=False
        )
        
        # 评估
        y_pred = self.model.predict(X_test)
        y_proba = self.model.predict_proba(X_test)
        accuracy = accuracy_score(y_test, y_pred)
        logloss = log_loss(y_test, y_proba)
        
        # 保存模型
        model_path = self.model_dir / "xgb_match_predictor.joblib"
        joblib.dump({"model": self.model, "features": list(X.columns)}, model_path)
        
        return {
            "accuracy": round(accuracy, 4),
            "log_loss": round(logloss, 4),
            "train_samples": len(X_train),
            "test_samples": len(X_test),
            "classification_report": classification_report(y_test, y_pred, 
                target_names=["主胜", "平局", "客胜"], output_dict=True),
            "model_path": str(model_path),
        }
    
    def cross_validate(self, X: pd.DataFrame, y: np.ndarray, n_splits: int = 5) -> dict:
        """时间序列交叉验证"""
        tscv = TimeSeriesSplit(n_splits=n_splits)
        scores = cross_val_score(self.model, X, y, cv=tscv, scoring="accuracy")
        return {"mean_accuracy": round(scores.mean(), 4), 
                "std_accuracy": round(scores.std(), 4),
                "fold_scores": scores.tolist()}
```

```python
# src/prediction/predictor.py
import joblib
import pandas as pd
import numpy as np
import shap

class MatchPredictor:
    """比赛预测器"""
    
    LABEL_MAP = {0: "主胜", 1: "平局", 2: "客胜"}
    
    def __init__(self, model_path: str):
        data = joblib.load(model_path)
        self.model = data["model"]
        self.feature_names = data["features"]
    
    def predict(self, features: dict) -> dict:
        """预测单场比赛"""
        X = pd.DataFrame([features])[self.feature_names]
        proba = self.model.predict_proba(X)[0]
        pred = int(np.argmax(proba))
        
        return {
            "prediction": self.LABEL_MAP[pred],
            "probabilities": {
                "home_win": round(float(proba[0]), 4),
                "draw": round(float(proba[1]), 4),
                "away_win": round(float(proba[2]), 4),
            },
            "confidence": round(float(max(proba)), 4),
            "features_used": features,
        }
    
    def explain(self, features: dict) -> dict:
        """SHAP可解释性分析：展示各特征对预测结果的贡献"""
        X = pd.DataFrame([features])[self.feature_names]
        explainer = shap.TreeExplainer(self.model)
        shap_values = explainer.shap_values(X)
        
        # 对预测类别提取特征贡献
        pred_class = int(np.argmax(self.model.predict_proba(X)[0]))
        contributions = dict(zip(self.feature_names, shap_values[pred_class][0]))
        
        # 按贡献绝对值排序
        sorted_contrib = sorted(contributions.items(), 
                               key=lambda x: abs(x[1]), reverse=True)
        
        return {
            "predicted_class": self.LABEL_MAP[pred_class],
            "base_value": float(explainer.expected_value[pred_class]),
            "top_positive_factors": [(k, round(float(v), 4)) 
                                      for k, v in sorted_contrib[:5] if v > 0],
            "top_negative_factors": [(k, round(float(v), 4)) 
                                      for k, v in sorted_contrib[:5] if v < 0],
            "all_contributions": {k: round(float(v), 4) 
                                  for k, v in sorted_contrib},
        }
```

## 十、前端设计（frontend/）

### 方案选择

| 方案 | 优点 | 缺点 | 适用场景 |
|---|---|---|---|
| **Streamlit** | 几行代码出交互页面，Python原生，开发快 | 定制化有限，性能一般 | 快速原型、答辩演示 |
| **Vue3 + ECharts** | 美观、高性能、完全定制 | 需要前端开发，工作量大 | 正式系统、长期维护 |
| **FastAPI + 模板引擎** | 轻量，不需要单独前端 | 交互性一般 | 简单展示 |

**建议**：毕业设计用 **Streamlit 做快速原型**（1-2天搞定），如果时间充裕再用 **Vue3 + ECharts** 做正式版。

### Streamlit 前端骨架

```python
# frontend/app.py
import streamlit as st
import requests
import pandas as pd

st.set_page_config(page_title="足球数据分析与AI预测", layout="wide")

API_BASE = "http://localhost:8000/api"

# 侧边栏导航
page = st.sidebar.radio("功能导航", 
    ["首页概览", "联赛数据", "球队分析", "球员分析", "比赛预测", "AI智能分析", "实时比分"])

if page == "比赛预测":
    st.title("比赛结果预测")
    
    col1, col2 = st.columns(2)
    with col1:
        home_team = st.selectbox("主队", ["阿森纳", "曼城", "利物浦", "切尔西"])
    with col2:
        away_team = st.selectbox("客队", ["阿森纳", "曼城", "利物浦", "切尔西"])
    
    if st.button("开始预测", type="primary"):
        with st.spinner("AI正在分析..."):
            resp = requests.post(f"{API_BASE}/predictions/predict",
                               json={"home_team": home_team, "away_team": away_team})
            result = resp.json()
            
            st.success(f"预测结果: {result['prediction']}")
            
            # 概率展示
            probs = result["probabilities"]
            cols = st.columns(3)
            cols[0].metric("主胜概率", f"{probs['home_win']:.1%}")
            cols[1].metric("平局概率", f"{probs['draw']:.1%}")
            cols[2].metric("客胜概率", f"{probs['away_win']:.1%}")
            
            # 可解释性
            st.subheader("关键影响因素")
            st.json(result.get("explanation", {}))

elif page == "AI智能分析":
    st.title("AI智能分析助手")
    st.markdown("可以问我任何关于球队、球员、比赛的问题")
    
    if "messages" not in st.session_state:
        st.session_state.messages = []
    
    for msg in st.session_state.messages:
        with st.chat_message(msg["role"]):
            st.markdown(msg["content"])
    
    if prompt := st.chat_input("输入你的问题..."):
        st.session_state.messages.append({"role": "user", "content": prompt})
        with st.chat_message("user"):
            st.markdown(prompt)
        
        with st.chat_message("assistant"):
            resp = requests.post(f"{API_BASE}/agent/chat", json={"message": prompt})
            answer = resp.json()["response"]
            st.markdown(answer)
        st.session_state.messages.append({"role": "assistant", "content": answer})
```

- "预测模型采用**40+维特征工程**，涵盖Elo实力、近期状态、主客场、历史交锋、攻防效率六大维度"

- "足球预测本身是难题（顶级模型准确率55-65%），本系统的价值在于**方法论和可解释性**，而非追求绝对准确率"

## 十五、风险与应对

| 风险 | 概率 | 影响 | 应对措施 |
|---|---|---|---|
| API免费额度不够 | 高 | 中 | 混合数据源，历史数据用CSV，实时用API；缓存策略减少重复请求 |
| 预测准确率低 | 高 | 中 | 不追求高准确率，强调方法论和可解释性；多模型对比展示 |
| Transfermarkt反爬 | 中 | 低 | 限速+随机UA+代理；实在不行用API-Football转会接口 |
| LLM API调用成本 | 中 | 低 | 用免费额度（DeepSeek新用户送额度）；缓存常见问题答案 |
| 时间不够做不完 | 中 | 高 | 按优先级开发，核心功能先完成，加分功能后补；Streamlit快速出原型 |
| 数据质量问题 | 中 | 中 | 数据清洗层做校验；缺失值处理；异常值检测 |
